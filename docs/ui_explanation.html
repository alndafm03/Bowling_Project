1. UIControls.js — لوحة التحكم

src/ui/UIControls.js

يبني لوحة التحكم (اليسار) ديناميكياً من مصفوفة الـ Sliders، ويُنظّم المفاتيح من لوحة المفاتيح. كل التفاعل يمر عبر Callbacks.

تعريف الـ Sliders

كل شريط تمرير مُعرَّف بكائن في مصفوفة SLIDERS:

// بنية كل Slider
{
  key:   'power',             // مفتاح للوصول من getControls()
  label: 'Power (V₀)',        // النص المعروض
  sub:   'launch speed',      // وصف صغير تحت الاسم
  min: 4, max: 12, step: 0.1, val: 8,  // نطاق القيمة
  fmt: (v) => `${v.toFixed(1)} m/s`    // دالة التنسيق
}

الـ Sliders السبعة وقيمها الافتراضية

Power (V₀) — سرعة الإطلاق
نطاق: 4–12 m/s | افتراضي: 8 m/s
التأثير: زيادة السرعة → مسافة انزلاق أطول، ضربة أقوى
Spin / Rev (ω) — الدوران الجانبي
نطاق: -25 إلى +25 rad/s | افتراضي: 10 rad/s
التأثير: القيم الموجبة تُميل الكرة يساراً (Hook)، السلبية يميناً
Lateral Start — موضع البداية الجانبي
نطاق: -0.45 إلى +0.45 m | افتراضي: 0 (المنتصف)
التأثير: يُغيّر نقطة إطلاق الكرة على عرض الممر
Aim Angle — زاوية التصويب
نطاق: -6° إلى +6° | افتراضي: 0°
التأثير: يُوجّه الكرة قليلاً يساراً أو يميناً عند الإطلاق
Ball Mass — كتلة الكرة
نطاق: 4–10 kg | افتراضي: 7 kg
التأثير: يُعيد بناء عزم القصور I = (2/5)mR² مباشرة
Ball Radius — نصف قطر الكرة
نطاق: 0.09–0.13 m | افتراضي: 0.11 m
التأثير: يُغيّر حجم الكرة المرئي والفيزيائي معاً
Lane Grip — قبضة الممر
نطاق: 0.5× إلى 1.8× | افتراضي: 1.0×
التأثير: يُقيّس كل معاملات الاحتكاك (زيتي ↔ جاف)

بناء الواجهة ديناميكياً

for (const s of SLIDERS) {
  // ننشئ: label > span + input[range] + output
  const wrap = document.createElement('label');
  const input = document.createElement('input');
  input.type = 'range';
  input.min = s.min; input.max = s.max;

  input.addEventListener('input', () => {
    out.textContent = s.fmt(+input.value);
    this._onInput(s.key); // يستدعي الـ callbacks المناسبة
  });
}

دالة getControls() — الأهم

تُعيد كل قيم الـ Sliders في كائن واحد، يُستخدم عند الإطلاق:

getControls() {
  return {
    power:    +this.inputs.power.value,    // V₀
    spin:     +this.inputs.spin.value,     // ω جانبي
    lateralX: +this.inputs.lateral.value,  // موضع X
    angleDeg: +this.inputs.angle.value,    // زاوية التصويب
    mass:     +this.inputs.mass.value,
    radius:   +this.inputs.radius.value,
    grip:     +this.inputs.grip.value,
  };
}

بناء صف واحد من HUD

_row(key, label, unit) {
  const row = document.createElement('div');
  row.className = 'hud-row';

  const s = document.createElement('span'); s.textContent = label;
  const b = document.createElement('b');    b.textContent = '0';
  const i = document.createElement('i');    i.textContent = unit;

  row.append(s, b, i);
  this.el[key] = b;  // نحفظ مرجع للقيمة لتحديثها لاحقاً
}

3. Renderer.js — المحرك البصري

يُهيئ Three.js WebGLRenderer، يُنشئ المشهد، ويُعدّ نظام الإضاءة الكامل.

إعدادات المُصيِّر

const renderer = new THREE.WebGLRenderer({
  antialias: true,                      // تنعيم الحواف
  powerPreference: 'high-performance'  // يطلب GPU المخصص
});
renderer.shadowMap.enabled = true;
renderer.shadowMap.type    = THREE.PCFSoftShadowMap;     // ظلال ناعمة
renderer.toneMapping       = THREE.ACESFilmicToneMapping; // تعيين الألوان سينمائي
renderer.toneMappingExposure = 1.05;

نظام الإضاءة (5 مصادر)

نوع الضوءاللونالوظيفة

4. CameraManager.js — أوضاع الكاميرا

src/rendering/CameraManager.js

يُدير 6 أوضاع كاميرا مختلفة، مع انتقالات ناعمة بين الأوضاع.

الأوضاع الستة

1. Orbit — دوران حر حول المشهد 1
2. Follow — تتبع الكرة من الخلف والأعلى 2
3. Top — مرئية من الأعلى (عصفور) 3
4. Side — رؤية جانبية للممر كاملاً 4
5. Impact — قريبة من منطقة الدبابيس 5
6. Free — تحكم حر باستخدام OrbitControls 6

الانتقالات الناعمة

الكاميرا لا تقفز فجأة — تُطير إلى الموضع الجديد بـ Lerp:

update(ballPos, dt) {
  // موضع الهدف حسب الوضع الحالي
  const target = this._getTargetPosition(ballPos);

  // انتقال ناعم — 8 × الفرق في كل إطار
  this.camera.position.lerp(target, Math.min(1, 8 * dt));
  this.camera.lookAt(lookAtPoint);
}

6. SimulationController.js — المنسّق الرئيسي

src/SimulationController.js

هذا الكلاس هو "المدير التنفيذي" للمشروع. يربط كل الأنظمة معاً ويُشغّل آلة الحالات.

كشف الوضع (Phase Detection)

classifyPhase() {
  if (!state.active)          return { label: 'Ready', cls: '' };
  if (ball.airborne)          return { label: 'Fall', cls: 'fall' };
  if (collisionFlash > 0)     return { label: 'Collision', cls: 'collision' };
  if (pinsMoving && ballNearDeck) return { label: 'Scatter', cls: 'scatter' };

  // كشف Skid/Hook/Roll من الانزلاق الفعلي
  if (slip > 0.3) {
    if (inDryZone && lateralSlip > 0.12 && sideSpin > 1)
      return { label: 'Hook', cls: 'hook' };
    return { label: 'Skid', cls: 'skid' };
  }
  return { label: 'Roll', cls: 'roll' };
}
مهم: الكشف يعتمد على قراءة الانزلاق الفعلي من الاصطدام (ليس مؤقتاً). هذا يعني أن الـ Phases ظاهرة طبيعية من الفيزياء، وليست مبرمجة يدوياً.

حساب التلميتري (Telemetry)

_pushTelemetry(dt) {
  const m = ball.mass, R = ball.radius;
  const v = ball.speed, w = ball.spin;
  const I = (2/5) * m * R * R;  // عزم القصور: I = ⅖mR²

  this.hud.setLive({
    speed:    v,
    angVel:   w,
    accel:    Δv / dt,           // |a| من تغير السرعة
    ke:       0.5 * m * v * v,   // الطاقة الانتقالية
    re:       0.5 * I * w * w,   // الطاقة الدورانية
    momentum: m * v,             // الزخم
    torque:   μ * m * G * R,     // العزم (أثناء الانزلاق)
  });
}

التسجيل — متى ينتهي الرمي؟

_resolveThrow(dt) {
  const ballDone = ball.isStopped() || ball.inGutter() || pastPins;

  if (ballDone && state.time > 0.8) {
    state.settle += dt;
    const pinsSettled = !pins.any(p => !p.sleeping);

    // ينتهي بعد 4 ثوانٍ أو 1.2 ثانية بعد سكون الدبابيس
    if (settle > 4 || (settle > 1.2 && pinsSettled)) {
      this._finish(pins.countDown());
    }
  }
}

Pin.js — الدبوس

كل دبوس: نموذج كبسولة (capsule) للاصطدام + RigidBody بتنسور اسطواني + مجموعة "كرات اتصال" للوقوف على الأرضية.

كيف يقف الدبوس؟ يُستخدم حلقة من 4 نقاط اتصال عند قاعدة الدبوس (دائرة support polygon). طالما مركز الكتلة يقع داخل هذه الدائرة، الدبوس يبقى واقفاً. أي ضربة تُخرجه → يسقط بشكل طبيعي.
